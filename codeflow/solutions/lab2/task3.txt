using System;

class Program {
    static void Main() {
        int i = int.Parse(Console.ReadLine());
        double d = double.Parse(Console.ReadLine());
        char ch = Console.ReadKey().KeyChar;
        Console.WriteLine();
        Console.WriteLine($"{i} ({i.GetType()}), {d} ({d.GetType()}), {ch} ({ch.GetType()})");
    }
}
