using System;
using System.Globalization;

class Program
{
    static void Main()
    {
        CultureInfo.CurrentCulture = CultureInfo.InvariantCulture;
        bool firstFlag = Convert.ToBoolean(int.Parse(Console.ReadLine()));
        bool secondFlag = Convert.ToBoolean(int.Parse(Console.ReadLine()));
        Console.WriteLine($"{firstFlag && secondFlag} {firstFlag || secondFlag} {firstFlag ^ secondFlag} {!firstFlag}");
    }
}
