using System;
using System.Globalization;

class Program
{
    static void Main()
    {
        CultureInfo.CurrentCulture = CultureInfo.InvariantCulture;
        int a = 5;
        double b = 3.14;
        char c = 'A';
        string s = "Hello";
        bool d = true;
        Console.WriteLine($"{a} {b} {c} {s} {d}");
    }
}
