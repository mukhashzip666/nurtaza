using System;
using System.Globalization;

class Program
{
    static void Main()
    {
        CultureInfo.CurrentCulture = CultureInfo.InvariantCulture;
        int integerValue = 5;
        double realValue = 3.14;
        char letter = 'A';
        string message = "Hello";
        bool flag = true;
        Console.WriteLine($"{integerValue} {realValue} {letter} {message} {flag}");
    }
}
