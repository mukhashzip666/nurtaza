using System;

class Program
{
    static void Main()
    {
        double a = Convert.ToDouble(Console.ReadLine());
        double b = Convert.ToDouble(Console.ReadLine());

        Console.WriteLine("Қосындысы: " + (a + b));
        Console.WriteLine("Айырмасы: " + (a - b));
        Console.WriteLine("Көбейтіндісі: " + (a * b));

        if (b != 0)
        {
            Console.WriteLine("Бөліндісі: " + (a / b));
        }
        else
        {
            Console.WriteLine("Нөлге бөлуге болмайды!");
        }
    }
}
