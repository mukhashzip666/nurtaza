using System;
using System.Globalization;

class Program
{
    static void Main()
    {
        CultureInfo.CurrentCulture = CultureInfo.InvariantCulture;
        double firstValue = Convert.ToDouble(Console.ReadLine());
        double secondValue = Convert.ToDouble(Console.ReadLine());
        Console.WriteLine("Қосындысы: " + (firstValue + secondValue));
        Console.WriteLine("Айырмасы: " + (firstValue - secondValue));
        Console.WriteLine("Көбейтіндісі: " + (firstValue * secondValue));
        if (secondValue != 0)
        {
            Console.WriteLine("Бөліндісі: " + (firstValue / secondValue));
        }
        else
        {
            Console.WriteLine("Нөлге бөлуге болмайды!");
        }
    }
}
