using System;
using System.Globalization;

class Program
{
    static void Main()
    {
        CultureInfo.CurrentCulture = CultureInfo.InvariantCulture;
        string s = Console.ReadLine();
        int a = int.Parse(Console.ReadLine());
        Console.WriteLine($"Your name is {s} and you are {a} years old.");
    }
}
