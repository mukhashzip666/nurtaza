using System;

class Program
{
    static void Main()
    {
        double a = double.Parse(Console.ReadLine());
        double b = double.Parse(Console.ReadLine());
        string op = Console.ReadLine();

        switch (op)
        {
            case "+":
                Console.WriteLine(a + b);
                break;
            case "-":
                Console.WriteLine(a - b);
                break;
            case "*":
                Console.WriteLine(a * b);
                break;
            case "/":
                if (b != 0)
                    Console.WriteLine(a / b);
                else
                    Console.WriteLine("Нөлге бөлуге болмайды!");
                break;
        }
    }
}
