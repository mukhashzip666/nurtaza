using System;
using System.Globalization;

class Program
{
    static void Main()
    {
        CultureInfo.CurrentCulture = CultureInfo.InvariantCulture;
        int rowCount = int.Parse(Console.ReadLine());
        for (long row = 1; row <= rowCount; row++)
        {
            for (long value = 1; value <= row; value++)
            {
                Console.Write(value);
            }

            Console.WriteLine();
        }
    }
}
