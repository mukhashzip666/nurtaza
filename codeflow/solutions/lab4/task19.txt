using System;

class Program
{
    static void Main()
    {
        int[,] a = new int[3, 3];

        for (int i = 0; i < 3; i++)
        {
            for (int j = 0; j < 3; j++)
            {
                a[i, j] = int.Parse(Console.ReadLine());
            }
        }

        for (int i = 0; i < 3; i++)
        {
            Console.Write(a[i, i] + " ");
        }
    }
}
