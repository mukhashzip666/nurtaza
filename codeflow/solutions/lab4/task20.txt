using System;
using System.Globalization;

class Program
{
    static void Main()
    {
        CultureInfo.CurrentCulture = CultureInfo.InvariantCulture;
        int[, ] matrix = new int[3, 3];
        for (int row = 0; row < 3; row++)
        {
            for (int column = 0; column < 3; column++)
            {
                matrix[row, column] = int.Parse(Console.ReadLine());
            }
        }

        long expectedSum = 0;
        for (int column = 0; column < 3; column++)
        {
            expectedSum += matrix[0, column];
        }

        bool isMagic = true;
        for (int row = 0; row < 3; row++)
        {
            long lineSum = 0;
            for (int column = 0; column < 3; column++)
            {
                lineSum += matrix[row, column];
            }

            if (lineSum != expectedSum)
            {
                isMagic = false;
            }
        }

        for (int column = 0; column < 3; column++)
        {
            long lineSum = 0;
            for (int row = 0; row < 3; row++)
            {
                lineSum += matrix[row, column];
            }

            if (lineSum != expectedSum)
            {
                isMagic = false;
            }
        }

        long mainDiagonalSum = 0;
        long secondaryDiagonalSum = 0;
        for (int row = 0; row < 3; row++)
        {
            mainDiagonalSum += matrix[row, row];
            secondaryDiagonalSum += matrix[row, 2 - row];
        }

        if (mainDiagonalSum != expectedSum || secondaryDiagonalSum != expectedSum)
        {
            isMagic = false;
        }

        if (isMagic)
        {
            Console.WriteLine("Magic square");
        }
        else
        {
            Console.WriteLine("Not a magic square");
        }
    }
}
