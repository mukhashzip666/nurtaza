using System;

class Program
{
    static void Main()
    {
        Console.Write("Enter number: ");
        int n = int.Parse(Console.ReadLine());

        bool prime = true;

        if (n < 2)
        {
            prime = false;
        }

        for (int i = 2; i < n; i++)
        {
            if (n % i == 0)
            {
                prime = false;
                break;
            }
        }

        if (prime)
            Console.WriteLine("Prime");
        else
            Console.WriteLine("Not prime");
    }
}
