using System;

class Program
{
    static void Main()
    {
        int[,] a = new int[3, 3];
        int sum = 0;

        for (int i = 0; i < 3; i++)
        {
            for (int j = 0; j < 3; j++)
            {
                a[i, j] = int.Parse(Console.ReadLine());
                sum += a[i, j];
            }
        }

        Console.WriteLine(sum);
    }
}
