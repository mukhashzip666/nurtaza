using System;
using System.Globalization;

class Program
{
    static void Main()
    {
        CultureInfo.CurrentCulture = CultureInfo.InvariantCulture;
        for (int row = 1; row <= 10; row++)
        {
            for (int column = 1; column <= 10; column++)
            {
                Console.Write(row * column + "\t");
            }

            Console.WriteLine();
        }
    }
}
