using System;
using System.Globalization;

class Program
{
    static void Main()
    {
        CultureInfo.CurrentCulture = CultureInfo.InvariantCulture;
        for (int i = 1; i <= 10; i++)
        {
            for (int j = 1; j <= 10; j++) Console.Write(i * j + "\t");
            Console.WriteLine();
        }
    }
}
