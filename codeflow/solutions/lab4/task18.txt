using System;
using System.Globalization;

class Program
{
    static void Main()
    {
        CultureInfo.CurrentCulture = CultureInfo.InvariantCulture;
        int rowCount = int.Parse(Console.ReadLine());
        for (int width = rowCount; width >= 1; width--)
        {
            for (int column = 1; column <= width; column++)
            {
                Console.Write("*");
            }

            Console.WriteLine();
        }
    }
}
