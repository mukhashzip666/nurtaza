using System;
using System.Globalization;

class Program
{
    static void Main()
    {
        CultureInfo.CurrentCulture = CultureInfo.InvariantCulture;
        int[] values = new int[10];
        for (int index = 0; index < values.Length; index++)
        {
            values[index] = int.Parse(Console.ReadLine());
        }

        long[] adjacentSums = new long[values.Length - 1];
        for (int index = 0; index < adjacentSums.Length; index++)
        {
            adjacentSums[index] = (long)values[index] + values[index + 1];
        }

        for (int index = 0; index < adjacentSums.Length; index++)
        {
            Console.WriteLine(adjacentSums[index]);
        }
    }
}
