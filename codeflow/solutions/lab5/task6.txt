using System;
using System.Globalization;

class Program
{
    static void Main()
    {
        CultureInfo.CurrentCulture = CultureInfo.InvariantCulture;
        int[] values = new int[10];
        int evenTotal = 0;
        int oddTotal = 0;
        for (int index = 0; index < values.Length; index++)
        {
            values[index] = int.Parse(Console.ReadLine());
            if (values[index] % 2 == 0)
            {
                evenTotal++;
            }
            else
            {
                oddTotal++;
            }
        }

        Console.WriteLine($"Жұп сандар саны: {evenTotal}");
        Console.WriteLine($"Тақ сандар саны: {oddTotal}");
    }
}
