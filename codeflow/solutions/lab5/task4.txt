using System;

class Program
{
    static void Main()
    {
        int[] numbers = new int[10];
        int sum = 0;

        for (int i = 0; i < numbers.Length; i++)
        {
            numbers[i] = int.Parse(Console.ReadLine());
            sum += numbers[i];
        }

        double average = (double)sum / numbers.Length;

        for (int i = 0; i < numbers.Length; i++)
        {
            if (numbers[i] > average)
            {
                Console.WriteLine(numbers[i]);
            }
        }
    }
}
