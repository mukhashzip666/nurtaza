using System;

class Program
{
    static void Main()
    {
        int[] numbers = new int[10];

        for (int i = 0; i < numbers.Length; i++)
        {
            numbers[i] = int.Parse(Console.ReadLine());
        }

        int target = int.Parse(Console.ReadLine());
        int index = -1;

        for (int i = 0; i < numbers.Length; i++)
        {
            if (numbers[i] == target)
            {
                index = i;
                break;
            }
        }

        if (index != -1)
        {
            Console.WriteLine($"Found it at index {index}!");
        }
        else
        {
            Console.WriteLine("Not here!");
        }
    }
}
