using System;
using System.Globalization;

class Program
{
    static void Main()
    {
        CultureInfo.CurrentCulture = CultureInfo.InvariantCulture;
        int[] values = new int[10];
        for (int index = 0; index < values.Length; index++)
        {
            values[index] = int.Parse(Console.ReadLine());
        }

        int maximumIndex = 0;
        int minimumIndex = 0;
        for (int index = 1; index < values.Length; index++)
        {
            if (values[index] > values[maximumIndex])
            {
                maximumIndex = index;
            }

            if (values[index] < values[minimumIndex])
            {
                minimumIndex = index;
            }
        }

        int temporary = values[maximumIndex];
        values[maximumIndex] = values[minimumIndex];
        values[minimumIndex] = temporary;
        for (int index = 0; index < values.Length; index++)
        {
            Console.WriteLine(values[index]);
        }
    }
}
