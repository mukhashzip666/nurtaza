using System;
using System.Globalization;

class Program
{
    static void Main()
    {
        CultureInfo.CurrentCulture = CultureInfo.InvariantCulture;
        Console.Write("Enter weigh (kg):");
        double weightKg = double.Parse(Console.ReadLine());
        Console.Write("Enter height (m):");
        double heightMeters = double.Parse(Console.ReadLine());
        double bodyMassIndex = weightKg / (heightMeters * heightMeters);
        Console.WriteLine("BMI: " + bodyMassIndex.ToString("F2"));
        if (bodyMassIndex < 18.5)
        {
            Console.WriteLine("Underweight");
        }
        else if (bodyMassIndex < 25)
        {
            Console.WriteLine("Normal");
        }
        else
        {
            Console.WriteLine("Overweight");
        }
    }
}
