using System;
using System.Globalization;

class Program
{
    static void Main()
    {
        CultureInfo.CurrentCulture = CultureInfo.InvariantCulture;
        Console.Write("Enter age: ");
        int driverAge = int.Parse(Console.ReadLine());
        if (driverAge >= 18)
        {
            Console.WriteLine("You can get a driving license");
        }
        else
        {
            Console.WriteLine("You cannot get a driving license");
        }
    }
}
