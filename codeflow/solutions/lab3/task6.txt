using System;
using System.Globalization;

class Program
{
    static void Main()
    {
        CultureInfo.CurrentCulture = CultureInfo.InvariantCulture;
        int k = I();
        string[] a = { "Pizza-$10", "Burger-$8", "Pasta-$9", "Salad-$6" };
        Console.WriteLine(k >= 1 && k <= 4 ? a[k - 1] : "Invalid choice");
    }

    static readonly System.IO.StreamReader Input = new System.IO.StreamReader(Console.OpenStandardInput());
    static string Next()
    {
        int c;
        do { c = Input.Read(); } while (c >= 0 && char.IsWhiteSpace((char)c));
        if (c < 0) throw new System.IO.EndOfStreamException();
        var s = new System.Text.StringBuilder();
        do { s.Append((char)c); c = Input.Read(); }
        while (c >= 0 && !char.IsWhiteSpace((char)c));
        return s.ToString();
    }
    static int I() { return int.Parse(Next(), CultureInfo.InvariantCulture); }
}
