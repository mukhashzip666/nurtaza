using System;
using System.Globalization;

class Program
{
    static void Main()
    {
        CultureInfo.CurrentCulture = CultureInfo.InvariantCulture;
        Console.Write("Enter menu choice (1-4):");
        int menuChoice = int.Parse(Console.ReadLine());
        switch (menuChoice)
        {
            case 1:
                Console.WriteLine("Pizza-$10");
                break;
            case 2:
                Console.WriteLine("Burger-$8");
                break;
            case 3:
                Console.WriteLine("Pasta-$9");
                break;
            case 4:
                Console.WriteLine("Salad-$6");
                break;
            default:
                Console.WriteLine("Invalid choice");
                break;
        }
    }
}
