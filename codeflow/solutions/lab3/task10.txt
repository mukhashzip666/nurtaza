using System;
using System.Globalization;

class Program
{
    static void Main()
    {
        CultureInfo.CurrentCulture = CultureInfo.InvariantCulture;
        Console.Write("Enter units used: ");
        double consumedUnits = double.Parse(Console.ReadLine());
        double totalCost;
        if (consumedUnits <= 100)
        {
            totalCost = consumedUnits * 0.5;
        }
        else if (consumedUnits <= 300)
        {
            totalCost = 100 * 0.5 + (consumedUnits - 100) * 0.75;
        }
        else
        {
            totalCost = 100 * 0.5 + 200 * 0.75 + (consumedUnits - 300) * 1.20;
        }

        Console.WriteLine("Total cost: " + totalCost);
    }
}
