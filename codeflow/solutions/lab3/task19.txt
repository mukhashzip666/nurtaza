using System;
using System.Globalization;

class Program
{
    static void Main()
    {
        CultureInfo.CurrentCulture = CultureInfo.InvariantCulture;
        Console.Write("Enter seat category: ");
        string seatCategory = Console.ReadLine();
        switch (seatCategory.ToLowerInvariant())
        {
            case "vip":
                Console.WriteLine("Price: 100");
                break;
            case "standard":
                Console.WriteLine("Price: 60");
                break;
            case "economy":
                Console.WriteLine("Price: 30");
                break;
            default:
                Console.WriteLine("Invalid category");
                break;
        }
    }
}
