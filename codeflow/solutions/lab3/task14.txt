using System;
using System.Globalization;

class Program
{
    static void Main()
    {
        CultureInfo.CurrentCulture = CultureInfo.InvariantCulture;
        int expectedPin = 1234;
        int enteredPin = int.Parse(Console.ReadLine());
        if (enteredPin == expectedPin)
        {
            Console.WriteLine("Снятие денег разрешено");
        }
        else
        {
            Console.WriteLine("Неверный PIN-код. Доступ запрещён");
        }
    }
}
