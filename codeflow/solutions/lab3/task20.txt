using System;
using System.Globalization;

class Program
{
    static void Main()
    {
        CultureInfo.CurrentCulture = CultureInfo.InvariantCulture;
        Console.Write("Enter number of days: ");
        int dayCount = int.Parse(Console.ReadLine());
        Console.Write("Enter normal price: ");
        double regularPrice = double.Parse(Console.ReadLine());
        double discountPercent;
        if (dayCount <= 3)
        {
            discountPercent = 0;
        }
        else if (dayCount <= 7)
        {
            discountPercent = 10;
        }
        else
        {
            discountPercent = 20;
        }

        double discountedPrice = regularPrice - regularPrice * discountPercent / 100;
        Console.WriteLine("Discount: " + discountPercent + "%");
        Console.WriteLine("Final price: " + discountedPrice);
    }
}
