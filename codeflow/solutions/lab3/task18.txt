using System;
using System.Globalization;

class Program
{
    static void Main()
    {
        CultureInfo.CurrentCulture = CultureInfo.InvariantCulture;
        Console.Write("Enter total classes: ");
        int lessonCount = int.Parse(Console.ReadLine());
        Console.Write("Enter attended classes: ");
        int visitedCount = int.Parse(Console.ReadLine());
        double attendancePercent = (double)visitedCount / lessonCount * 100;
        if (attendancePercent >= 75)
        {
            Console.WriteLine("Eligible");
        }
        else
        {
            Console.WriteLine("Not eligible");
        }
    }
}
