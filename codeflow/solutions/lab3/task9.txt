using System;
using System.Globalization;

class Program
{
    static void Main()
    {
        CultureInfo.CurrentCulture = CultureInfo.InvariantCulture;
        Console.Write("Enter usage in GB: ");
        double usedGigabytes = double.Parse(Console.ReadLine());
        if (usedGigabytes < 5)
        {
            Console.WriteLine("Basic Plan");
        }
        else if (usedGigabytes <= 15)
        {
            Console.WriteLine("Standard Plan");
        }
        else
        {
            Console.WriteLine("Premium Plan");
        }
    }
}
