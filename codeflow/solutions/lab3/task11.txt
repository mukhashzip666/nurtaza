using System;
using System.Globalization;

class Program
{
    static void Main()
    {
        CultureInfo.CurrentCulture = CultureInfo.InvariantCulture;
        Console.Write("Enter marks: ");
        int score = int.Parse(Console.ReadLine());
        if (score >= 90 && score <= 100)
        {
            Console.WriteLine("A");
        }
        else if (score >= 75)
        {
            Console.WriteLine("B");
        }
        else if (score >= 50)
        {
            Console.WriteLine("C");
        }
        else
        {
            Console.WriteLine("Fail");
        }
    }
}
