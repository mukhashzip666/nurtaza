using System;
using System.Globalization;

class Program
{
    static void Main()
    {
        CultureInfo.CurrentCulture = CultureInfo.InvariantCulture;
        Console.Write("Enter cart total: ");
        double cartTotal = double.Parse(Console.ReadLine());
        double discountPercent;
        if (cartTotal > 100)
        {
            discountPercent = 20;
        }
        else if (cartTotal > 50)
        {
            discountPercent = 10;
        }
        else
        {
            discountPercent = 0;
        }

        double amountToPay = cartTotal - cartTotal * discountPercent / 100;
        Console.WriteLine("Discount: " + discountPercent + "%");
        Console.WriteLine("Final price: " + amountToPay);
    }
}
