Ввод чисел, операций и отдельных слов — через любые пробельные
разделители: пробелы, табуляции, переносы строк. Имена, логин и пароли
читаются по строкам там, где это указано. Приглашения ко вводу убраны,
исходные подписи результатов сохранены. Все вспомогательные методы
включены в соответствующую программу.

Вещественные числа вводятся с точкой. Размер массива или матрицы
перед элементами вводить не нужно: размер задан в коде.

long; для факториала и Фибоначчи — BigInteger. Подробнее — в коде
и примечаниях к заданиям. Ввод символа в 2.3 работает через stdin.

ВАРИАНТ 1
========================================================================
Ввод: два целых числа.

```csharp
using System;
using System.Globalization;

class Program
{
    static void Main()
    {
        CultureInfo.CurrentCulture = CultureInfo.InvariantCulture;
        int a = I(), b = I();
        Console.WriteLine((long)a + b);
    }

    static readonly System.IO.StreamReader Input = new System.IO.StreamReader(Console.OpenStandardInput());
    static string Next()
    {
        int c;
        do { c = Input.Read(); } while (c >= 0 && char.IsWhiteSpace((char)c));
        if (c < 0) throw new System.IO.EndOfStreamException();
        var s = new System.Text.StringBuilder();
        do { s.Append((char)c); c = Input.Read(); }
        while (c >= 0 && !char.IsWhiteSpace((char)c));
        return s.ToString();
    }
    static int I() { return int.Parse(Next(), CultureInfo.InvariantCulture); }
}
```

Калькулятор
ВАРИАНТ 1
========================================================================
Ввод: два вещественных числа, затем знак +, -, * или /.
Примечание: Порядок ввода: a b знак, а не a знак b. Неизвестная операция, как в исходнике, ничего не выводит.

```csharp
using System;
using System.Globalization;

class Program
{
    static void Main()
    {
        CultureInfo.CurrentCulture = CultureInfo.InvariantCulture;
        double a = D(), b = D();
        string op = Next();
        switch (op)
        {
            case "+": Console.WriteLine(a + b); break;
            case "-": Console.WriteLine(a - b); break;
            case "*": Console.WriteLine(a * b); break;
            case "/":
                if (b != 0) Console.WriteLine(a / b);
                else Console.WriteLine("Нөлге бөлуге болмайды!");
                break;
        }
    }

    static readonly System.IO.StreamReader Input = new System.IO.StreamReader(Console.OpenStandardInput());
    static string Next()
    {
        int c;
        do { c = Input.Read(); } while (c >= 0 && char.IsWhiteSpace((char)c));
        if (c < 0) throw new System.IO.EndOfStreamException();
        var s = new System.Text.StringBuilder();
        do { s.Append((char)c); c = Input.Read(); }
        while (c >= 0 && !char.IsWhiteSpace((char)c));
        return s.ToString();
    }
    static double D() { return double.Parse(Next(), CultureInfo.InvariantCulture); }
}
```

    }

    static readonly System.IO.StreamReader Input = new System.IO.StreamReader(Console.OpenStandardInput());
    static string Next()
    {
        int c;

        if (c < 0) throw new System.IO.EndOfStreamException();
        var s = new System.Text.StringBuilder();
        do { s.Append((char)c); c = Input.Read(); }
        while (c >= 0 && !char.IsWhiteSpace((char)c));
        return s.ToString();
    }
    static int I() { return int.Parse(Next(), CultureInfo.InvariantCulture); }
    static double D() { return double.Parse(Next(), CultureInfo.InvariantCulture); }
}

ЛАБОРАТОРНАЯ №4 — ЗАДАНИЕ №3
Сумма от 1 до n
ВАРИАНТ 1
========================================================================
Ввод: целое n.
Примечание: Вариант 1 использует формулу за O(1), вариант 2 сохраняет цикл O(n). При n <= 0 оба выводят 0, как исходник.

```csharp
using System;
using System.Globalization;

class Program
{
    static void Main()
    {
        CultureInfo.CurrentCulture = CultureInfo.InvariantCulture;
        long n = I();
        Console.WriteLine(n > 0 ? n * (n + 1) / 2 : 0);
    }

    static readonly System.IO.StreamReader Input = new System.IO.StreamReader(Console.OpenStandardInput());
    static string Next()
    {
        int c;
        do { c = Input.Read(); } while (c >= 0 && char.IsWhiteSpace((char)c));
        if (c < 0) throw new System.IO.EndOfStreamException();
        var s = new System.Text.StringBuilder();
        do { s.Append((char)c); c = Input.Read(); }
        while (c >= 0 && !char.IsWhiteSpace((char)c));
        return s.ToString();
    }
    static int I() { return int.Parse(Next(), CultureInfo.InvariantCulture); }
}
```
